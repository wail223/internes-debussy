<script>

    const internes = ['Waël', 'Noussaiba', 'Yassamine', 'Lylia', 'Ikram', 'Baya'];

    // Date de référence des gardes

    const baseDateUTC = Date.UTC(2026, 8, 29); 

    // Dimanche de référence pour déterminer quelle semaine on est

    const refSundayUTC = Date.UTC(2026, 8, 27);


    document.addEventListener("DOMContentLoaded", () => {

        let today = new Date();

        document.getElementById('startDate').value = today.toISOString().split('T')[0];

        document.getElementById('interneFiltre').addEventListener('change', genererPlanning);

        genererPlanning();

    });


    function genererPlanning() {

        const startDateInput = document.getElementById('startDate').value;

        if (!startDateInput) return;

        

        const weeks = parseInt(document.getElementById('weeksCount').value);

        const daysToGenerate = weeks * 7;

        const targetInterne = parseInt(document.getElementById('interneFiltre').value);

        

        const startParts = startDateInput.split('-');

        let currentUTC = Date.UTC(startParts[0], startParts[1] - 1, startParts[2]);

        

        let tbody = document.getElementById('tableBody');

        tbody.innerHTML = ''; 

        

        const joursSemaine = ['Dimanche', 'Lundi', 'Mardi', 'Mercredi', 'Jeudi', 'Vendredi', 'Samedi'];


        for (let i = 0; i < daysToGenerate; i++) {

            let loopDate = new Date(currentUTC + i * 86400000);

            let diffDays = Math.floor((loopDate.getTime() - baseDateUTC) / 86400000);

            let diffFromSunday = Math.floor((loopDate.getTime() - refSundayUTC) / 86400000);

            

            // Alternance des semaines pour les groupes

            let currentWeekIndex = Math.floor(diffFromSunday / 7);

            let isGroupeA = (currentWeekIndex % 2 === 0);

            

            // LE HACK MATHEMATIQUE : Groupes Impairs (A) et Pairs (B)

            // Groupe A : 0 (Waël), 2 (Yassamine), 4 (Ikram)

            // Groupe B : 1 (Noussaiba), 3 (Lylia), 5 (Baya)

            let groupeActif = isGroupeA ? [0, 2, 4] : [1, 3, 5];

            

            if (loopDate.getUTCDay() === 0 && i !== 0) {

                let sepTr = document.createElement('tr');

                sepTr.innerHTML = `<td colspan="4" class="semaine-separator">🔽 SEMAINE DU GROUPE ${isGroupeA ? 'A (Waël, Yassamine, Ikram)' : 'B (Noussaiba, Lylia, Baya)'} 🔽</td>`;

                tbody.appendChild(sepTr);

            } else if (i === 0) {

                let sepTr = document.createElement('tr');

                sepTr.innerHTML = `<td colspan="4" class="semaine-separator">SEMAINE EN COURS : GROUPE ${isGroupeA ? 'A (Waël, Yassamine, Ikram)' : 'B (Noussaiba, Lylia, Baya)'}</td>`;

                tbody.appendChild(sepTr);

            }

            

            let gardeIdx = ((diffDays % 6) + 6) % 6;

            let postGardeIdx = (gardeIdx - 1 + 6) % 6;

            

            let dayOfWeek = loopDate.getUTCDay();

            let isWeekend = (dayOfWeek === 5 || dayOfWeek === 6);

            

            let presents = [];

            if (!isWeekend) {

                // Les présents sont le groupe de la semaine MOINS l'interne de garde et l'interne post-garde

                // (Mathématiquement, ça donnera TOUJOURS exactement 2 présents !)

                presents = groupeActif.filter(idx => idx !== gardeIdx && idx !== postGardeIdx);

            }

            

            let repos = [];

            for (let j = 0; j < 6; j++) {

                if (j !== gardeIdx && !presents.includes(j)) {

                    repos.push(j);

                }

            }


            let rowClass = "";

            if (isWeekend) rowClass = "weekend-row";


            if (targetInterne !== -1) {

                if (targetInterne === gardeIdx) rowClass += " hl-garde";

                else if (presents.includes(targetInterne)) rowClass += " hl-present";

                else if (repos.includes(targetInterne)) rowClass += " hl-repos";

            }


            let tr = document.createElement('tr');

            tr.className = rowClass.trim();


            let dateStr = `${joursSemaine[dayOfWeek]} ${loopDate.getUTCDate().toString().padStart(2, '0')}/${(loopDate.getUTCMonth()+1).toString().padStart(2, '0')}`;

            

            const formatName = (idx, baseClass = "") => {

                let name = internes[idx];

                if (idx === targetInterne) return `<span class="${baseClass} highlight-text">${name}</span>`;

                return `<span class="${baseClass}">${name}</span>`;

            };


            let gardeStr = `<span class="badge-garde">${internes[gardeIdx]}</span>`;

            if (targetInterne === gardeIdx) gardeStr = `<span class="badge-garde highlight-text">${internes[gardeIdx]}</span>`;


            let presentsStr = isWeekend ? "<em>- Aucun -</em>" : presents.map(idx => formatName(idx)).join(', ');

            

            let reposStr = repos.map(idx => {

                let isPostGarde = (idx === postGardeIdx);

                let cssClass = isPostGarde ? "post-garde-text" : "";

                let nameHtml = formatName(idx, cssClass);

                return isPostGarde ? `${nameHtml} (Post-garde)` : nameHtml;

            }).join(', ');


            tr.innerHTML = `

                <td><strong>${dateStr}</strong></td>

                <td>${gardeStr}</td>

                <td>${presentsStr}</td>

                <td>${reposStr}</td>

            `;

            

            tbody.appendChild(tr);

        }

    }

</script>
