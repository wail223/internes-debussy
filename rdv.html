<script>

    function calculerRDV() {

        let targetDay = parseInt(document.getElementById('jourMedecin').value);

        let delai = parseInt(document.getElementById('delai').value);

        

        let dateRDV = new Date();

        dateRDV.setDate(dateRDV.getDate() + delai);

        

        // Ajuster au jour de la semaine ciblé

        while (dateRDV.getDay() !== targetDay) {

            dateRDV.setDate(dateRDV.getDate() + 1);

        }


        let options = { weekday: 'long', year: 'numeric', month: 'long', day: 'numeric' };

        document.getElementById('resultat').innerText = "Prochain RDV :\n" + dateRDV.toLocaleDateString('fr-FR', options);

    }

</script>
